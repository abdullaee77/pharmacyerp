class MedicineId extends ValueObject {
  final String value;
  const MedicineId(this.value);

  factory MedicineId.generate() {
    final ts = DateTime.now().millisecondsSinceEpoch;
    final rand = Random().nextInt(99999).toString().padLeft(5, '0');
    return MedicineId('med_${ts}_$rand');
  }

  @override
  bool operator ==(Object other) =>
      identical(this, other) ||
          other is MedicineId && runtimeType == other.runtimeType && value == other.value;

  @override
  int get hashCode => value.hashCode;

  @override
  String toString() => value;
}

    return purchasePrice;

/// Money value object using integer minor units (paisa) to avoid
/// floating-point rounding errors in financial calculations.
///
/// 1 PKR = 100 paisa.
/// PKR 150.50 â†’ paisa 15050.
class Money extends ValueObject {
  /// Amount in minor units (paisa).
  final int paisa;

  const Money({required this.paisa});

  /// Create from a PKR double value. Use only for UI input conversion.
  factory Money.fromPkr(double pkr) => Money(paisa: (pkr * 100).round());

  /// Create directly from integer paisa (preferred for storage/domain).
  factory Money.fromPaisa(int paisa) => Money(paisa: paisa);

  /// Zero amount.
  factory Money.zero() => const Money(paisa: 0);

  /// Convert to PKR double for display only.
  double get pkr => paisa / 100;

  /// Formatted display string.
  String get display => 'PKR ${pkr.toStringAsFixed(2)}';

  bool get isZero => paisa == 0;
  bool get isPositive => paisa > 0;

  Money operator +(Money other) => Money(paisa: paisa + other.paisa);
  Money operator -(Money other) => Money(paisa: paisa - other.paisa);

  @override
  bool operator ==(Object other) =>
      identical(this, other) ||
          other is Money && runtimeType == other.runtimeType && paisa == other.paisa;

  @override
  int get hashCode => paisa.hashCode;

  @override
  String toString() => display;
}

/// Barcode value object with domain validation.
class Barcode extends ValueObject {
  final String value;

  const Barcode._(this.value);

  /// Creates a Barcode after validation.
  /// Returns null if the barcode is empty or invalid.
  static Barcode? create(String raw) {
    final trimmed = raw.trim();
    if (trimmed.isEmpty) return null;
    if (trimmed.length > 50) return null;
    return Barcode._(trimmed);
  }

  /// Creates a Barcode without validation (for known-good data from DB).
  factory Barcode.unsafe(String value) => Barcode._(value);

  @override
  bool operator ==(Object other) =>
      identical(this, other) ||
          other is Barcode && runtimeType == other.runtimeType && value == other.value;

  @override
  int get hashCode => value.hashCode;

  @override
  String toString() => value;
}

/// Quantity value object ensuring non-negative inventory counts.
class Quantity extends ValueObject {
  final int value;

  const Quantity._(this.value);

  /// Creates a Quantity, clamping to zero if negative.
  factory Quantity.create(int amount) =>
      Quantity._(amount < 0 ? 0 : amount);

  factory Quantity.zero() => const Quantity._(0);

  bool get isZero => value == 0;
  bool get isLow => value > 0 && value <= 10;

  @override
  bool operator ==(Object other) =>
      identical(this, other) ||
          other is Quantity && runtimeType == other.runtimeType && value == other.value;

  @override
  int get hashCode => value.hashCode;

  @override
  String toString() => '$value';
}

class _ParsedRow {
  final int rowNumber;
  final Medicine? medicine;
  final String? error;
  final Map<String, String> raw;

  _ParsedRow.ok(this.rowNumber, Medicine this.medicine, this.raw) : error = null;
  _ParsedRow.err(this.rowNumber, this.error, this.raw) : medicine = null;

  bool get isOk => medicine != null;
}

class MedicineBulkImportDialog extends StatefulWidget {
  const MedicineBulkImportDialog({super.key});

  @override
  State<MedicineBulkImportDialog> createState() =>
      _MedicineBulkImportDialogState();
}

class _MedicineBulkImportDialogState extends State<MedicineBulkImportDialog> {
  List<_ParsedRow> _rows = [];
  String? _fileName;
  bool _loading = false;
  String? _error;

  // Expected CSV/Excel headers (case-insensitive, flexible)
  static const _headers = [
    'name',
    'generic_name',
    'manufacturer',
    'dosage_form',
    'strength',
    'category',
    'unit',
    'rack_location',
    'purchase_price',
    'selling_price',
    'mrp',
    'box_size',
    'min_stock',
    'barcode',
    'prescription_required',
  ];

  Future<void> _pickFile() async {
    setState(() {
      _loading = true;
      _error = null;
      _rows = [];
    });

    try {
      final result = await FilePicker.platform.pickFiles(
        type: FileType.custom,
        allowedExtensions: ['csv', 'xlsx', 'xls'],
      );
      if (result == null || result.files.single.path == null) {
        setState(() => _loading = false);
        return;
      }

      final path = result.files.single.path!;
      final name = result.files.single.name;
      _fileName = name;

      List<List<String>> table;
      if (name.toLowerCase().endsWith('.csv')) {
        table = await _readCsv(path);
      } else {
        table = await _readExcel(path);
      }

      if (table.isEmpty) {
        setState(() {
          _error = 'File is empty.';
          _loading = false;
        });
        return;
      }

      final headerRow =
      table.first.map((e) => e.trim().toLowerCase()).toList();
      final dataRows = table.skip(1).toList();

      final parsed = <_ParsedRow>[];
      for (var i = 0; i < dataRows.length; i++) {
        final rowNum = i + 2; // +1 for header, +1 to make human-friendly
        final raw = <String, String>{};
        for (var c = 0; c < headerRow.length && c < dataRows[i].length; c++) {
          raw[headerRow[c]] = dataRows[i][c].trim();
        }
        if (raw.values.every((v) => v.isEmpty)) continue; // skip blank line

        try {
          final med = _buildMedicine(raw);
          parsed.add(_ParsedRow.ok(rowNum, med, raw));
        } catch (e) {
          parsed.add(_ParsedRow.err(rowNum, e.toString(), raw));
        }
      }

      setState(() {
        _rows = parsed;
        _loading = false;
      });
    } catch (e) {
      setState(() {
        _error = 'Failed to read file: $e';
        _loading = false;
      });
    }
  }

  Future<List<List<String>>> _readCsv(String path) async {
    final content = await File(path).readAsString();
    final raw = const CsvToListConverter(
      eol: '\n',
      shouldParseNumbers: false,
    ).convert(content);
    return raw
        .map((r) => r.map((c) => (c ?? '').toString()).toList())
        .toList();
  }

  Future<List<List<String>>> _readExcel(String path) async {
    final bytes = await File(path).readAsBytes();
    final excel = Excel.decodeBytes(bytes);
    if (excel.tables.isEmpty) return [];
    final sheet = excel.tables[excel.tables.keys.first]!;
    final out = <List<String>>[];
    for (final row in sheet.rows) {
      out.add(row
          .map((c) => c?.value?.toString().trim() ?? '')
          .toList());
    }
    return out;
  }

  Medicine _buildMedicine(Map<String, String> row) {
    final name = row['name'] ?? '';
    if (name.isEmpty) {
      throw 'Column "name" is required';
    }

    final purchase = double.tryParse(row['purchase_price'] ?? '') ?? 0.0;
    final selling = double.tryParse(row['selling_price'] ?? '') ?? 0.0;
    final mrp = double.tryParse(row['mrp'] ?? '') ?? selling;
    final boxSize = int.tryParse(row['box_size'] ?? '') ?? 1;
    final minStock = int.tryParse(row['min_stock'] ?? '') ?? 0;
    final rxReq = _parseBool(row['prescription_required']);

    final now = DateTime.now();
    return Medicine(
      id: MedicineId.generate(),
      name: name,
      genericName: row['generic_name'] ?? '',
      manufacturer: row['manufacturer'] ?? '',
      dosageForm: _parseDosage(row['dosage_form']),
      strength: row['strength'] ?? '',
      category: row['category'] ?? '',
      unit: (row['unit']?.isNotEmpty ?? false) ? row['unit']! : 'Tab',
      rackLocation: row['rack_location'] ?? '',
      drugSchedule: DrugSchedule.none,
      storageInstructions: '',
      barcode: (row['barcode']?.isNotEmpty ?? false)
          ? Barcode.unsafe(row['barcode']!)
          : null,
      prescriptionRequired: rxReq,
      minStockLevel: Quantity.create(minStock),
      purchasePrice: Money.fromPkr(purchase),
      sellingPrice: Money.fromPkr(selling),
      mrp: Money.fromPkr(mrp),
      boxSize: boxSize > 0 ? boxSize : 1,
      boxPrice: Money.fromPkr(selling * (boxSize > 0 ? boxSize : 1)),
      status: MedicineStatus.active,
      createdAt: now,
      updatedAt: now,
    );
  }

  bool _parseBool(String? v) {
    if (v == null) return false;
    final s = v.trim().toLowerCase();
    return s == '1' || s == 'true' || s == 'yes' || s == 'y';
  }

  DosageForm _parseDosage(String? v) {
    if (v == null || v.isEmpty) return DosageForm.tablet;
    final s = v.trim().toLowerCase();
    return DosageForm.values.firstWhere(
          (e) => e.name.toLowerCase() == s || e.label.toLowerCase() == s,
      orElse: () => DosageForm.other,
    );
  }

import 'package:pharmacy/core/result/result.dart';
import 'package:pharmacy/core/error/failures.dart';
import '../domain/medicine.dart';

class CreateMedicineUseCase {
  final MedicineRepository _repository;
  const CreateMedicineUseCase(this._repository);

  Future<Result<Medicine>> execute(Medicine medicine) async {
    if (medicine.name.trim().isEmpty) {
      return const Failure(ValidationFailure(message: 'Medicine name is required.'));
    }
    if (medicine.name.trim().length < 2) {
      return const Failure(ValidationFailure(message: 'Medicine name must be at least 2 characters.'));
    }
    if (medicine.sellingPrice.paisa < 0) {
      return const Failure(ValidationFailure(message: 'Selling price cannot be negative.'));
    }
    if (medicine.purchasePrice.paisa < 0) {
      return const Failure(ValidationFailure(message: 'Purchase price cannot be negative.'));
    }
    if (medicine.mrp.paisa < 0) {
      return const Failure(ValidationFailure(message: 'MRP cannot be negative.'));
    }
    if (!medicine.isPriceValid) {
      return const Failure(ValidationFailure(message: 'Selling price cannot exceed MRP.'));
    }
    if (medicine.boxSize < 1) {
      return const Failure(ValidationFailure(message: 'Box size must be at least 1.'));
    }
    return _repository.createMedicine(medicine);
  }
}

class UpdateMedicineUseCase {
  final MedicineRepository _repository;
  const UpdateMedicineUseCase(this._repository);

  Future<Result<Medicine>> execute(Medicine medicine) async {
    if (medicine.name.trim().isEmpty) {
      return const Failure(ValidationFailure(message: 'Medicine name is required.'));
    }
    if (!medicine.isPriceValid) {
      return const Failure(ValidationFailure(message: 'Selling price cannot exceed MRP.'));
    }
    return _repository.updateMedicine(medicine);
  }
}
