class PurchaseFormDialog extends StatefulWidget {
  final PurchaseController controller;

  const PurchaseFormDialog({super.key, required this.controller});

  @override
  State<PurchaseFormDialog> createState() => _PurchaseFormDialogState();
}

class _PurchaseFormDialogState extends State<PurchaseFormDialog> {
  final _formKey = GlobalKey<FormState>();

  final SqliteMedicineRepository _medicineRepo = SqliteMedicineRepository();
  final SqliteSupplierRepository _supplierRepo = SqliteSupplierRepository();

  // Header controllers
  final _invoiceNumCtrl = TextEditingController();
  final _supplierSearchCtrl = TextEditingController();
  final _discountCtrl = TextEditingController(text: '0');

  // Active line-item controllers
  final _medSearchCtrl = TextEditingController();
  final _batchCtrl = TextEditingController();
  final _qtyCtrl = TextEditingController();
  final _purPriceCtrl = TextEditingController();
  final _selPriceCtrl = TextEditingController();

  DateTime _expiryDate = DateTime.now().add(const Duration(days: 365));

  // Selection & State
  Medicine? _selectedMedicine;
  List<Medicine> _searchResults = [];
  bool _isSearchingMed = false;

  List<SupplierWithBalance> _suppliers = [];
  String _selectedSupplierName = 'General Supplier';

  final List<PurchaseItem> _addedItems = [];
  bool _isSaving = false;

  @override
  void initState() {
    super.initState();
    _invoiceNumCtrl.text = PurchaseInvoiceNumber.generate().value;
    _loadSuppliers();
  }

  @override
  void dispose() {
    _invoiceNumCtrl.dispose();
    _supplierSearchCtrl.dispose();
    _discountCtrl.dispose();
    _medSearchCtrl.dispose();
    _batchCtrl.dispose();
    _qtyCtrl.dispose();
    _purPriceCtrl.dispose();
    _selPriceCtrl.dispose();
    super.dispose();
  }

  Future<void> _loadSuppliers() async {
    final result = await _supplierRepo.getSuppliers();
    result.fold(
      onSuccess: (list) {
        if (mounted) {
          setState(() {
            _suppliers = list;
            if (list.isNotEmpty) {
              _selectedSupplierName = list.first.supplier.name;
            }
          });
        }
      },
      onFailure: (_) {},
    );
  }

  Future<void> _searchMedicines(String query) async {
    if (query.trim().isEmpty) {
      setState(() {
        _searchResults = [];
        _isSearchingMed = false;
      });
      return;
    }

    setState(() => _isSearchingMed = true);

    final res = await _medicineRepo.getMedicines(
      filter: MedicineFilter(searchQuery: query.trim()),
    );

    res.fold(
      onSuccess: (list) {
        if (mounted) {
          setState(() {
            _searchResults = list;
            _isSearchingMed = false;
          });
        }
      },
      onFailure: (_) {
        if (mounted) setState(() => _isSearchingMed = false);
      },
    );
  }

  void _selectMedicine(Medicine med) {
    setState(() {
      _selectedMedicine = med;
      _medSearchCtrl.text = '${med.name} (${med.strength})';
      _searchResults = [];
      _purPriceCtrl.text = med.effectivePurchasePrice.pkr > 0
          ? med.effectivePurchasePrice.pkr.toStringAsFixed(2)
          : '';
      _selPriceCtrl.text = med.effectiveSellingPrice.pkr > 0
          ? med.effectiveSellingPrice.pkr.toStringAsFixed(2)
          : '';
    });
  }

  void _clearMedicineSelection() {
    setState(() {
      _selectedMedicine = null;
      _medSearchCtrl.clear();
      _searchResults = [];
      _purPriceCtrl.clear();
      _selPriceCtrl.clear();
    });
  }

  Future<void> _pickExpiry() async {
    final picked = await showDatePicker(
      context: context,
      initialDate: _expiryDate,
      firstDate: DateTime.now(),
      lastDate: DateTime(2040),
    );
    if (picked != null) {
      setState(() => _expiryDate = picked);
    }
  }

  void _addItemToInvoice() {
    if (_selectedMedicine == null) {
      AppToast.error(context, 'Please search and select a medicine from catalog.');
      return;
    }

    final batchNum = _batchCtrl.text.trim();
    if (batchNum.isEmpty) {
      AppToast.error(context, 'Please enter a batch number.');
      return;
    }

    final qty = int.tryParse(_qtyCtrl.text.trim()) ?? 0;
    if (qty <= 0) {
      AppToast.error(context, 'Procured quantity must be greater than 0.');
      return;
    }

    final purCostVal = double.tryParse(_purPriceCtrl.text.trim()) ?? 0.0;
    final selPriceVal = double.tryParse(_selPriceCtrl.text.trim()) ?? 0.0;

    final purchasePrice = Money.fromPkr(purCostVal);
    final sellingPrice = Money.fromPkr(selPriceVal);
    final lineTotal = Money.fromPaisa(purchasePrice.paisa * qty);

    final item = PurchaseItem(
      id: 'p_item_${DateTime.now().microsecondsSinceEpoch}',
      medicineId: _selectedMedicine!.id, // LINKED TO REAL CATALOG MEDICINE ID
      medicineName: _selectedMedicine!.name,
      batchNumber: batchNum,
      expiryDate: _expiryDate,
      quantity: qty,
      purchasePrice: purchasePrice,
      sellingPrice: sellingPrice,
      lineTotal: lineTotal,
    );

    setState(() {
      _addedItems.add(item);
      _clearMedicineSelection();
      _batchCtrl.clear();
      _qtyCtrl.clear();
    });
  }

  void _removeItem(int index) {
    setState(() => _addedItems.removeAt(index));
  }

  int get _subtotalPaisa =>
      _addedItems.fold(0, (sum, item) => sum + item.lineTotal.paisa);

  int get _discountPaisa {
    final val = double.tryParse(_discountCtrl.text.trim()) ?? 0.0;
    return Money.fromPkr(val).paisa;
  }

  int get _grandTotalPaisa {
    final total = _subtotalPaisa - _discountPaisa;
    return total < 0 ? 0 : total;
  }

  Future<void> _submitInvoice() async {
    if (_addedItems.isEmpty) {
      AppToast.error(context, 'Please add at least one medicine item to invoice.');
      return;
    }

    final supplier = _selectedSupplierName.trim();
    if (supplier.isEmpty) {
      AppToast.error(context, 'Supplier name cannot be empty.');
      return;
    }

    setState(() => _isSaving = true);

    final now = DateTime.now();
    final pId = PurchaseId.generate();
    final invNum = _invoiceNumCtrl.text.trim().isNotEmpty
        ? PurchaseInvoiceNumber(_invoiceNumCtrl.text.trim())
        : PurchaseInvoiceNumber.generate();

    final purchase = Purchase(
      id: pId,
      invoiceNumber: invNum,
      supplierName: supplier,
      items: List.from(_addedItems),
      subtotal: Money.fromPaisa(_subtotalPaisa),
      discount: Money.fromPaisa(_discountPaisa),
      grandTotal: Money.fromPaisa(_grandTotalPaisa),
      createdAt: now,
      updatedAt: now,
    );

    final error = await widget.controller.receivePurchase(purchase);

    setState(() => _isSaving = false);
    if (mounted) Navigator.of(context).pop(error);
  }

                  IconButton(
                    icon: const Icon(Icons.close_rounded, size: 20),

                  ),
                ],
              ),
            ),
            const Divider(height: 1),

            Expanded(

import 'package:flutter/material.dart';
import '../../../inventory/domain/batch_repository.dart';

class PurchaseController extends ChangeNotifier {
  final ReceivePurchaseUseCase _receive;
  final GetPurchasesUseCase _getPurchases;
  final CreatePurchaseReturnUseCase _createReturn;
  final PurchaseRepository _purchaseRepository;

  PurchaseController({
    required PurchaseRepository purchaseRepository,
    required InventoryRepository inventoryRepository,
    required BatchRepository batchRepository,
    SupplierRepository? supplierRepository,
  })  : _purchaseRepository = purchaseRepository,
        _receive = ReceivePurchaseUseCase(
          purchaseRepository: purchaseRepository,
          inventoryRepository: inventoryRepository,
          batchRepository: batchRepository,
          supplierRepository: supplierRepository,
        ),
        _getPurchases = GetPurchasesUseCase(purchaseRepository),
        _createReturn = CreatePurchaseReturnUseCase(
          purchaseRepository: purchaseRepository,
          inventoryRepository: inventoryRepository,
        );

  List<Purchase> _purchases = [];
  bool _isLoading = false;
  String? _error;
  String _searchQuery = '';

  List<Purchase> get purchases => _purchases;
  bool get isLoading => _isLoading;
  String? get error => _error;
  String get searchQuery => _searchQuery;

  Future<void> loadPurchases({bool silent = false}) async {
    if (!silent) {
      _isLoading = true;
      _error = null;
      notifyListeners();
    }

    final result = await _getPurchases.execute(
        searchQuery: _searchQuery.isEmpty ? null : _searchQuery);
    result.fold(
      onSuccess: (data) {
        _purchases = data;
        _error = null;
        _isLoading = false;
        notifyListeners();
      },
      onFailure: (f) {
        _error = f.message;
        _isLoading = false;
        notifyListeners();
      },
    );
  }

  void search(String query) {
    _searchQuery = query;
    loadPurchases();
  }

  Future<String?> receivePurchase(Purchase purchase) async {
    final result = await _receive.execute(purchase);
    return result.fold(
      onSuccess: (_) {
        loadPurchases();
        return null;
      },
      onFailure: (f) => f.message,
    );
  }

  Future<String?> createReturn({
    required Purchase originalPurchase,
    required PurchaseReturn returnDoc,
  }) async {
    final result = await _createReturn.execute(
        originalPurchase: originalPurchase, returnDoc: returnDoc);
    return result.fold(
      onSuccess: (_) {
        loadPurchases();
        return null;
      },
      onFailure: (f) => f.message,
    );
  }

  Future<Map<String, int>> loadReturnedQuantities(PurchaseId id) async {
    final result = await _purchaseRepository.getReturnedQuantities(id);
    return result.fold(
      onSuccess: (d) => d,
      onFailure: (_) => {},
    );
  }
}
